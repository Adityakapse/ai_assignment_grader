        int s = 0;

            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    s += a[i] + a[j] + a[k];
                }

    }

        int n = a.length;
        int t = 0;
        for (int i = 0; i < n; i++) {
            t += a[i];
        }
        return t;
    }
}
