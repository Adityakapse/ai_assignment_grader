def f(mat):
    n = len(mat)
    s = 0
    for i in range(n):
        s += mat[i][0]
    t = g(mat)
    return s + t

def g(mat):
    n = len(mat)
    t = 0
    for i in range(n):
        for j in range(n):
            t += mat[i][j]
    return t
