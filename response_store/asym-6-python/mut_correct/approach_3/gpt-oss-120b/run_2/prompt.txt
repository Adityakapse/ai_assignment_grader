def f(a):
    n = len(a)
    s = 0
    for i in range(n):
        s += a[i]
    t = g(a)
    return s + t

def g(a):
    n = len(a)
    t = 0
    k = n
    while k > 0:
        t += a[k - 1]
        k = k // 2
    return t
