public final class AsymQ19 {
    // `a` has length N; `b` is sorted ascending and also has length N.
    public static int f(int[] a, int[] b) {
        int n = a.length;
        int s = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                s += a[j];
            }
        }
        int t = g(b, a[0]);
        return s + t;
    }

    private static int g(int[] b, int x) {
        int l = 0;
        int r = b.length - 1;
        while (l <= r) {
            int mid = (l + r) / 2;
            if (b[mid] == x) return mid;
            if (b[mid] < x) l = mid + 1;
            else r = mid - 1;
        }
        return -1;
    }
}
