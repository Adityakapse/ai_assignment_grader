        int n = mat.length;
        int c = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (g(b, mat[i][j]) >= 0) {
                    c++;
                }
            }
        }

            else r = mid - 1;
        }
        return -1;
    }
