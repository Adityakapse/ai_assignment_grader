private static void merge(int[] array, int[] tmp, int f1, int t1, int f2, int t2) {
    int i = f1; // Index in the original array
    int j = f2; // Index in the original array
    int k = f1; // Index in the temporary array

    // Merge the two subarrays into the temporary array
    while (k <= t2) {
        if (i <= t1 && (j > t2 || array[i] <= array[j])) {
            tmp[k] = array[i];
            i++;
        } else {
            tmp[k] = array[j];
            j++;
        }
        k++;
    }

    // Copy the merged elements back to the original array
    for (k = f1; k <= t2; k++) {
        array[k] = tmp[k];
    }
}
