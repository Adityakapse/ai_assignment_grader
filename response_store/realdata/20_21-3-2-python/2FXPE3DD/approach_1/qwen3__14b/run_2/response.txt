def longest_path_dag(graph, nodes):

    virtual_node = 'virtual'
    graph[virtual_node] = list(nodes)
    nodes = [virtual_node] + nodes

    # Step 2: Compute in-degrees for topological sorting
    in_degree = defaultdict(int)
    for u in graph:
        for v in graph[u]:
            in_degree[v] += 1

    # Step 3: Topological sort
    queue = deque([node for node in nodes if in_degree[node] == 0])
    topological_order = []

    while queue:
        u = queue.popleft()
        topological_order.append(u)
        for v in graph[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)

    # Step 4: Dynamic programming to compute longest path
    dp = {node: 0 for node in nodes}
    for u in topological_order:
        for v in graph[u]:
            dp[v] = max(dp[v], dp[u] + 1)

    return max(dp.values())
